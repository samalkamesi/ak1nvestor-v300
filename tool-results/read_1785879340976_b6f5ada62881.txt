    60→/** Map a variable id (e.g. "V01") to its deep course slug. */
    61→function variableIdToSlug(vid: string): string | null {
    62→  const slug = Object.keys(deepCourses).find((s) =>
    63→    s.startsWith(vid.toLowerCase().replace("v", "v").substring(0, 3))
    64→  );
    65→  return slug || null;
    66→}
