  1613→# Helper: build the chapter payload (preserving num, minutes, title)

  1616→def build_course_payload(course, new_data):
  1617→    """Return a new course dict with replaced fields, preserving
  1618→    chapter num/minutes/title."""
  1619→    new_course = deepcopy(course)
  1620→    new_course["why"] = new_data["why"]
  1621→    new_course["history"] = deepcopy(new_data["history"])
  1622→    new_course["lynchSection"] = new_data["lynchSection"]
  1623→    new_course["grahamSection"] = new_data["grahamSection"]
  1624→    new_course["ak1Section"] = new_data["ak1Section"]
  1625→
  1626→    # Replace chapters while keeping num/minutes/title from chapters_list
  1627→    existing_chapters = course.get("chapters", [])
  1628→    chapter_list = course.get("chapters_list", [])
  1629→    new_chapters = []
  1630→    for i, ch_data in enumerate(new_data["chapters"]):
  1631→        # Prefer the metadata from existing chapter if available; fallback to chapters_list
  1632→        if i < len(existing_chapters):
  1633→            num = existing_chapters[i].get("num", i + 1)
  1634→            minutes = existing_chapters[i].get("minutes", 4)
  1635→            title = existing_chapters[i].get("title", f"{i+1}. Kapitel")
  1636→        elif i < len(chapter_list):
  1637→            num = chapter_list[i].get("num", i + 1)
  1638→            minutes = chapter_list[i].get("minutes", 4)
  1639→            title = chapter_list[i].get("title", f"{i+1}. Kapitel")
  1640→        else:
  1641→            num = i + 1
  1642→            minutes = 4
  1643→            title = f"{i+1}. Kapitel"
  1644→
  1645→        new_chapters.append({
  1646→            "num": num,
  1647→            "minutes": minutes,
  1648→            "title": title,
  1649→            "intro": ch_data["intro"],
  1650→            "blocks": deepcopy(ch_data["blocks"]),
  1651→        })
  1652→    new_course["chapters"] = new_chapters
  1653→    return new_course
