   507→function getLevel(xp: number): { level: number; title: string; nextAt: number } {
   508→  if (xp < 500) return { level: 1, title: "NYANALYTIKER", nextAt: 500 };
   509→  if (xp < 1500) return { level: 2, title: "ANALYTIKER-ELEV", nextAt: 1500 };
   510→  if (xp < 4000) return { level: 3, title: "ANALYTIKER", nextAt: 4000 };
   511→  if (xp < 9000) return { level: 4, title: "SENIOR ANALYTIKER", nextAt: 9000 };
   512→  return { level: 5, title: "MASTER ANALYTIKER", nextAt: -1 };
   513→}
   514→
   515→function verdict(score: number): { label: string; color: string } {
   516→  if (score > 75) return { label: "Stark", color: "text-bull" };
   517→  if (score >= 50) return { label: "Bra", color: "text-gold" };
   518→  return { label: "Svag", color: "text-bear" };
   519→}
