   112→function timeAgo(iso: string): string {
   113→  const diff = Date.now() - new Date(iso).getTime();
   114→  const sec = Math.floor(diff / 1000);
   115→  if (sec < 60) return `${sec}s sedan`;
   116→  const min = Math.floor(sec / 60);
   117→  if (min < 60) return `${min}m sedan`;
   118→  const hr = Math.floor(min / 60);
   119→  if (hr < 24) return `${hr}h sedan`;
   120→  const day = Math.floor(hr / 24);
   121→  return `${day}d sedan`;
   122→}
